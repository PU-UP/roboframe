cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RoboframeStdio LANGUAGES CXX)
    include(CTest)
endif()
if(NOT UNIX)
    message(FATAL_ERROR "StdioChannel requires a POSIX platform")
endif()

find_package(Threads REQUIRED)
add_library(roboframe_stdio_transport stdio_channel.cpp)
target_link_libraries(roboframe_stdio_transport PUBLIC Threads::Threads)
target_include_directories(roboframe_stdio_transport PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(roboframe_stdio_transport PUBLIC cxx_std_14)

if(BUILD_TESTING)
    if(NOT TARGET Roboframe::sdk)
        find_package(RoboframeSDK CONFIG REQUIRED)
    endif()
    add_executable(stdio_channel_test tests/stdio_channel_test.cpp)
    target_link_libraries(stdio_channel_test PRIVATE roboframe_stdio_transport Roboframe::sdk)
    add_test(NAME stdio_channel_test COMMAND stdio_channel_test)
    set_tests_properties(stdio_channel_test PROPERTIES TIMEOUT 10)
endif()

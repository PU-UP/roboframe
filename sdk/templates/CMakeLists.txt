cmake_minimum_required(VERSION 3.16)
project(RoboframeTemplates LANGUAGES CXX)
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()

add_executable(stream_component_check component/stream_check.cpp)
add_executable(goal_component_check component/goal_check.cpp)
add_executable(stream_integration_check integration/stream_check.cpp)
add_executable(bt_integration_check integration/bt_check.cpp)
foreach(target stream_component_check goal_component_check stream_integration_check bt_integration_check)
    target_link_libraries(${target} PRIVATE Roboframe::sdk)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    if(BUILD_TESTING)
        add_test(NAME ${target} COMMAND ${target})
        set_tests_properties(${target} PROPERTIES TIMEOUT 10)
    endif()
endforeach()
target_link_libraries(bt_integration_check PRIVATE Roboframe::orchestration)

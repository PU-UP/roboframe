cmake_minimum_required(VERSION 3.16)
project(RoboframeStarter LANGUAGES CXX)
include(CTest)
add_subdirectory(component)
add_executable(integration_demo integration/main.cpp)
target_link_libraries(integration_demo PRIVATE ValueScale::component)
set_target_properties(integration_demo PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_test(NAME integration_demo COMMAND integration_demo)
endif()

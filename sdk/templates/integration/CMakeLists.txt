cmake_minimum_required(VERSION 3.16)
project(RoboframeIntegrationTemplate LANGUAGES CXX)
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
add_executable(integration_template_check check.cpp)
target_link_libraries(integration_template_check PRIVATE Roboframe::sdk)
set_target_properties(integration_template_check PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_test(NAME integration_template_check COMMAND integration_template_check)
    set_tests_properties(integration_template_check PROPERTIES TIMEOUT 10)
endif()

add_executable(server_template_check server_check.cpp)
target_link_libraries(server_template_check PRIVATE Roboframe::orchestration)
set_target_properties(server_template_check PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_test(NAME server_template_check COMMAND server_template_check)
    set_tests_properties(server_template_check PROPERTIES TIMEOUT 10)
endif()

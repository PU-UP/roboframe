cmake_minimum_required(VERSION 3.16)
project(ValueScale VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK 1.0 CONFIG REQUIRED)
endif()

add_library(value_scale STATIC src/component.cpp)
add_library(ValueScale::component ALIAS value_scale)
target_include_directories(value_scale PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(value_scale PUBLIC Roboframe::sdk)
target_compile_definitions(value_scale PRIVATE VALUE_SCALE_VERSION="${PROJECT_VERSION}")
add_executable(value_scale_demo examples/main.cpp)
target_link_libraries(value_scale_demo PRIVATE ValueScale::component)
if(BUILD_TESTING)
    add_test(NAME value_scale_demo COMMAND value_scale_demo)
endif()

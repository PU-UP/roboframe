cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
include("${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(MapUpdater VERSION ${component_version} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK ${component_sdk_min_version} CONFIG REQUIRED)
endif()
add_library(map_updater STATIC src/occupancy_map_component.cpp)
add_library(MapUpdater::component ALIAS map_updater)
set_target_properties(map_updater PROPERTIES EXPORT_NAME component)
target_compile_features(map_updater PUBLIC cxx_std_14)
target_include_directories(map_updater PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(map_updater PUBLIC Roboframe::sdk)
roboframe_component_metadata(map_updater map_updater)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(map_updater PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(occupancy_map_test tests/occupancy_map_component_test.cpp)
    target_link_libraries(occupancy_map_test PRIVATE MapUpdater::component)
    add_test(NAME occupancy_map_test COMMAND occupancy_map_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/MapUpdater)
configure_package_config_file(cmake/MapUpdaterConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/MapUpdaterConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MapUpdaterConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(TARGETS map_updater EXPORT MapUpdaterTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MapUpdaterTargets NAMESPACE MapUpdater:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MapUpdaterConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/MapUpdaterConfigVersion.cmake DESTINATION ${package_dir})

cmake_minimum_required(VERSION 3.16)
project(Slam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(slam STATIC src/recorded_slam_component.cpp)
add_library(Slam::component ALIAS slam)
set_target_properties(slam PROPERTIES EXPORT_NAME component)
target_compile_features(slam PUBLIC cxx_std_14)
target_include_directories(slam PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(slam PUBLIC Roboframe::sdk)
roboframe_component_metadata(slam slam)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(slam PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(recorded_slam_test tests/recorded_slam_component_test.cpp)
    target_link_libraries(recorded_slam_test PRIVATE Slam::component)
    add_test(NAME recorded_slam_test COMMAND recorded_slam_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/Slam)
configure_package_config_file(cmake/SlamConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/SlamConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS slam EXPORT SlamTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SlamTargets NAMESPACE Slam:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SlamConfig.cmake DESTINATION ${package_dir})

cmake_minimum_required(VERSION 3.16)
project(PoseGraph LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(posegraph STATIC src/pose_graph_component.cpp)
add_library(PoseGraph::component ALIAS posegraph)
set_target_properties(posegraph PROPERTIES EXPORT_NAME component)
target_compile_features(posegraph PUBLIC cxx_std_14)
target_include_directories(posegraph PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(posegraph PUBLIC Roboframe::sdk)
roboframe_component_metadata(posegraph posegraph)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(posegraph PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(pose_graph_component_test tests/pose_graph_component_test.cpp)
    target_link_libraries(pose_graph_component_test PRIVATE PoseGraph::component)
    add_test(NAME pose_graph_component_test COMMAND pose_graph_component_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/PoseGraph)
configure_package_config_file(cmake/PoseGraphConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/PoseGraphConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS posegraph EXPORT PoseGraphTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PoseGraphTargets NAMESPACE PoseGraph:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PoseGraphConfig.cmake DESTINATION ${package_dir})

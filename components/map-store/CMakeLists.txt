cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS VERSION component_version LIMIT_COUNT 1)
project(MapStore VERSION ${component_version} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK 1.1 CONFIG REQUIRED)
endif()
add_library(map_store STATIC src/component.cpp)
add_library(MapStore::component ALIAS map_store)
set_target_properties(map_store PROPERTIES EXPORT_NAME component)
target_compile_features(map_store PUBLIC cxx_std_14)
target_compile_definitions(map_store PRIVATE ROBOFRAME_COMPONENT_VERSION="${PROJECT_VERSION}")
target_include_directories(map_store PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(map_store PUBLIC Roboframe::sdk)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(map_store PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(map_store_check tests/check.cpp)
    target_link_libraries(map_store_check PRIVATE MapStore::component)
    add_test(NAME map_store_check COMMAND map_store_check)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/MapStore)
configure_package_config_file(cmake/MapStoreConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/MapStoreConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MapStoreConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(TARGETS map_store EXPORT MapStoreTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MapStoreTargets NAMESPACE MapStore:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MapStoreConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/MapStoreConfigVersion.cmake DESTINATION ${package_dir})

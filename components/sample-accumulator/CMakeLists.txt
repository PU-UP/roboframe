cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS VERSION SAMPLE_ACCUMULATOR_VERSION LIMIT_COUNT 1)
project(SampleAccumulator VERSION ${SAMPLE_ACCUMULATOR_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK 1.0 CONFIG REQUIRED)
endif()
option(SAMPLE_ACCUMULATOR_BUILD_DEMO "Build the direct component example" ON)

add_library(sample_accumulator STATIC src/component.cpp)
add_library(SampleAccumulator::component ALIAS sample_accumulator)
set_target_properties(sample_accumulator PROPERTIES EXPORT_NAME component)
target_compile_features(sample_accumulator PUBLIC cxx_std_14)
target_link_libraries(sample_accumulator PUBLIC Roboframe::sdk)
target_compile_definitions(sample_accumulator PRIVATE SAMPLE_ACCUMULATOR_VERSION="${PROJECT_VERSION}")
target_include_directories(sample_accumulator PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sample_accumulator PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(SAMPLE_ACCUMULATOR_BUILD_DEMO)
    add_executable(accumulator_demo examples/main.cpp)
    target_link_libraries(accumulator_demo PRIVATE SampleAccumulator::component)
endif()
if(BUILD_TESTING)
    add_executable(component_check tests/component_check.cpp)
    target_link_libraries(component_check PRIVATE SampleAccumulator::component)
    add_test(NAME component_check COMMAND component_check)
    if(SAMPLE_ACCUMULATOR_BUILD_DEMO)
        add_test(NAME accumulator_demo COMMAND accumulator_demo)
    endif()
endif()

set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/SampleAccumulator)
configure_package_config_file(cmake/SampleAccumulatorConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/SampleAccumulatorConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SampleAccumulatorConfigVersion.cmake
    COMPATIBILITY SameMajorVersion)
install(TARGETS sample_accumulator EXPORT SampleAccumulatorTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SampleAccumulatorTargets NAMESPACE SampleAccumulator:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SampleAccumulatorConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/SampleAccumulatorConfigVersion.cmake DESTINATION ${package_dir})

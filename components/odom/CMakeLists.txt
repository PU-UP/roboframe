cmake_minimum_required(VERSION 3.16)
project(Odom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()

add_library(odom STATIC src/odom_component.cpp)
add_library(Odom::component ALIAS odom)
set_target_properties(odom PROPERTIES EXPORT_NAME component)
target_compile_features(odom PUBLIC cxx_std_14)
target_include_directories(odom PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(odom PUBLIC Roboframe::sdk)
roboframe_component_metadata(odom odom)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(odom PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(BUILD_TESTING)
    add_executable(odom_component_test tests/odom_component_test.cpp)
    target_link_libraries(odom_component_test PRIVATE Odom::component)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(odom_component_test PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
    add_test(NAME odom_component_test COMMAND odom_component_test)
    set_tests_properties(odom_component_test PROPERTIES TIMEOUT 10)
endif()

option(ODOM_BUILD_RECEIVER "Build the standalone odometry receiver" ON)
if(ODOM_BUILD_RECEIVER)
    if(NOT TARGET roboframe_stdio_transport)
        set(interface_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../interface")
        if(NOT EXISTS "${interface_dir}/CMakeLists.txt")
            message(FATAL_ERROR "ODOM_BUILD_RECEIVER needs interface/; use OFF for a library-only build")
        endif()
        add_subdirectory("${interface_dir}" "${CMAKE_CURRENT_BINARY_DIR}/interface")
    endif()
    add_executable(odom_receive main.cpp)
    target_link_libraries(odom_receive PRIVATE Odom::component roboframe_stdio_transport)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(odom_receive PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
    if(BUILD_TESTING)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        set(odom_log "${CMAKE_CURRENT_SOURCE_DIR}/../../data/intel-lab/intel.log.gz")
        if(NOT EXISTS "${odom_log}")
            message(FATAL_ERROR "Odometry replay tests require data/intel-lab/intel.log.gz")
        endif()
        set(replay_check "${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_test.py")
        add_test(NAME odom_replay COMMAND "${Python3_EXECUTABLE}" "${replay_check}"
            $<TARGET_FILE:odom_receive> "${odom_log}")
        add_test(NAME odom_replay_corrupt_gzip COMMAND "${Python3_EXECUTABLE}" "${replay_check}"
            $<TARGET_FILE:odom_receive> "${odom_log}" --check-corrupt)
        set_tests_properties(odom_replay odom_replay_corrupt_gzip PROPERTIES TIMEOUT 60)
    endif()
endif()

set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/Odom)
configure_package_config_file(cmake/OdomConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/OdomConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS odom EXPORT OdomTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT OdomTargets NAMESPACE Odom:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OdomConfig.cmake DESTINATION ${package_dir})

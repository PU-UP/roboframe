cmake_minimum_required(VERSION 3.16)
project(Odom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(odom STATIC src/passthrough_odom_component.cpp)
add_library(Odom::component ALIAS odom)
set_target_properties(odom PROPERTIES EXPORT_NAME component)
target_compile_features(odom PUBLIC cxx_std_14)
target_include_directories(odom PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(odom PUBLIC Roboframe::sdk)
roboframe_component_metadata(odom odom)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(odom PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(passthrough_odom_test tests/passthrough_odom_component_test.cpp)
    target_link_libraries(passthrough_odom_test PRIVATE Odom::component)
    add_test(NAME passthrough_odom_test COMMAND passthrough_odom_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/Odom)
configure_package_config_file(cmake/OdomConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/OdomConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS odom EXPORT OdomTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT OdomTargets NAMESPACE Odom:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OdomConfig.cmake DESTINATION ${package_dir})

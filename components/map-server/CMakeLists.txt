cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
include("${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(MapServer VERSION ${component_version} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK ${component_sdk_min_version} CONFIG REQUIRED)
endif()
add_library(map_server STATIC src/file_map_component.cpp)
add_library(MapServer::component ALIAS map_server)
set_target_properties(map_server PROPERTIES EXPORT_NAME component)
target_compile_features(map_server PUBLIC cxx_std_14)
target_include_directories(map_server PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(map_server PUBLIC Roboframe::sdk)
roboframe_component_metadata(map_server map_server)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(map_server PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(file_map_test tests/file_map_component_test.cpp)
    target_link_libraries(file_map_test PRIVATE MapServer::component)
    add_test(NAME file_map_test COMMAND file_map_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/MapServer)
configure_package_config_file(cmake/MapServerConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/MapServerConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MapServerConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(TARGETS map_server EXPORT MapServerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MapServerTargets NAMESPACE MapServer:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MapServerConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/MapServerConfigVersion.cmake DESTINATION ${package_dir})

cmake_minimum_required(VERSION 3.16)
project(Planner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(planner STATIC src/read_only_planner_component.cpp)
add_library(Planner::component ALIAS planner)
set_target_properties(planner PROPERTIES EXPORT_NAME component)
target_compile_features(planner PUBLIC cxx_std_14)
target_include_directories(planner PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(planner PUBLIC Roboframe::sdk)
roboframe_component_metadata(planner planner)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(planner PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(read_only_planner_test tests/read_only_planner_component_test.cpp)
    target_link_libraries(read_only_planner_test PRIVATE Planner::component)
    add_test(NAME read_only_planner_test COMMAND read_only_planner_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/Planner)
configure_package_config_file(cmake/PlannerConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/PlannerConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS planner EXPORT PlannerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PlannerTargets NAMESPACE Planner:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PlannerConfig.cmake DESTINATION ${package_dir})

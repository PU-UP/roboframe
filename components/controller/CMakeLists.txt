cmake_minimum_required(VERSION 3.16)
project(Controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(controller STATIC src/controller_component.cpp)
add_library(Controller::component ALIAS controller)
set_target_properties(controller PROPERTIES EXPORT_NAME component)
target_compile_features(controller PUBLIC cxx_std_14)
target_include_directories(controller PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(controller PUBLIC Roboframe::sdk)
roboframe_component_metadata(controller controller)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(controller_execution_test tests/controller_execution_test.cpp)
    target_link_libraries(controller_execution_test PRIVATE Controller::component)
    add_test(NAME controller_execution_test COMMAND controller_execution_test)
    set_tests_properties(controller_execution_test PROPERTIES TIMEOUT 10)
    add_executable(controller_algorithm_test tests/controller_algorithm_test.cpp)
    target_link_libraries(controller_algorithm_test PRIVATE Controller::component)
    add_test(NAME controller_algorithm_test COMMAND controller_algorithm_test)
    set_tests_properties(controller_algorithm_test PROPERTIES TIMEOUT 10)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/Controller)
configure_package_config_file(cmake/ControllerConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/ControllerConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS controller EXPORT ControllerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ControllerTargets NAMESPACE Controller:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ControllerConfig.cmake DESTINATION ${package_dir})

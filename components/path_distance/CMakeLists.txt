cmake_minimum_required(VERSION 3.16)
project(PathDistance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()

add_library(path_distance STATIC src/path_distance_component.cpp)
add_library(PathDistance::component ALIAS path_distance)
set_target_properties(path_distance PROPERTIES EXPORT_NAME component)
target_compile_features(path_distance PUBLIC cxx_std_14)
target_include_directories(path_distance PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(path_distance PUBLIC Roboframe::sdk)
roboframe_component_metadata(path_distance path_distance)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(path_distance PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(BUILD_TESTING)
    add_executable(path_distance_component_test tests/path_distance_component_test.cpp)
    target_link_libraries(path_distance_component_test PRIVATE PathDistance::component)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(path_distance_component_test PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
    add_test(NAME path_distance_component_test COMMAND path_distance_component_test)
    set_tests_properties(path_distance_component_test PROPERTIES TIMEOUT 10)
endif()

set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/PathDistance)
configure_package_config_file(cmake/PathDistanceConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/PathDistanceConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS path_distance EXPORT PathDistanceTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PathDistanceTargets NAMESPACE PathDistance:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PathDistanceConfig.cmake DESTINATION ${package_dir})

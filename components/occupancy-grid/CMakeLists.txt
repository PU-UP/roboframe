cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS VERSION component_version LIMIT_COUNT 1)
project(OccupancyGrid VERSION ${component_version} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK 1.1 CONFIG REQUIRED)
endif()
add_library(occupancy_grid STATIC src/component.cpp)
add_library(OccupancyGrid::component ALIAS occupancy_grid)
set_target_properties(occupancy_grid PROPERTIES EXPORT_NAME component)
target_compile_features(occupancy_grid PUBLIC cxx_std_14)
target_compile_definitions(occupancy_grid PRIVATE ROBOFRAME_COMPONENT_VERSION="${PROJECT_VERSION}")
target_include_directories(occupancy_grid PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(occupancy_grid PUBLIC Roboframe::sdk)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(occupancy_grid PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(occupancy_grid_check tests/check.cpp)
    target_link_libraries(occupancy_grid_check PRIVATE OccupancyGrid::component)
    add_test(NAME occupancy_grid_check COMMAND occupancy_grid_check)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/OccupancyGrid)
configure_package_config_file(cmake/OccupancyGridConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/OccupancyGridConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OccupancyGridConfigVersion.cmake COMPATIBILITY SameMajorVersion)
install(TARGETS occupancy_grid EXPORT OccupancyGridTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT OccupancyGridTargets NAMESPACE OccupancyGrid:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OccupancyGridConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/OccupancyGridConfigVersion.cmake DESTINATION ${package_dir})

cmake_minimum_required(VERSION 3.16)
project(LocalMapBuilder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(localmapbuilder STATIC src/local_map_component.cpp)
add_library(LocalMapBuilder::component ALIAS localmapbuilder)
set_target_properties(localmapbuilder PROPERTIES EXPORT_NAME component)
target_compile_features(localmapbuilder PUBLIC cxx_std_14)
target_include_directories(localmapbuilder PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(localmapbuilder PUBLIC Roboframe::sdk)
roboframe_component_metadata(localmapbuilder localmapbuilder)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(localmapbuilder PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(local_map_component_test tests/local_map_component_test.cpp)
    target_link_libraries(local_map_component_test PRIVATE LocalMapBuilder::component)
    add_test(NAME local_map_component_test COMMAND local_map_component_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/LocalMapBuilder)
configure_package_config_file(cmake/LocalMapBuilderConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/LocalMapBuilderConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS localmapbuilder EXPORT LocalMapBuilderTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT LocalMapBuilderTargets NAMESPACE LocalMapBuilder:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LocalMapBuilderConfig.cmake DESTINATION ${package_dir})

cmake_minimum_required(VERSION 3.16)
project(DistanceGoal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()

add_library(distance_goal STATIC src/distance_goal_component.cpp)
add_library(DistanceGoal::component ALIAS distance_goal)
set_target_properties(distance_goal PROPERTIES EXPORT_NAME component)
target_compile_features(distance_goal PUBLIC cxx_std_14)
target_include_directories(distance_goal PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(distance_goal PUBLIC Roboframe::sdk)
roboframe_component_metadata(distance_goal distance_goal)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(distance_goal PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(BUILD_TESTING)
    add_executable(distance_goal_component_test tests/distance_goal_component_test.cpp)
    target_link_libraries(distance_goal_component_test PRIVATE DistanceGoal::component)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(distance_goal_component_test PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
    add_test(NAME distance_goal_component_test COMMAND distance_goal_component_test)
    set_tests_properties(distance_goal_component_test PROPERTIES TIMEOUT 10)
endif()

set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/DistanceGoal)
configure_package_config_file(cmake/DistanceGoalConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/DistanceGoalConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS distance_goal EXPORT DistanceGoalTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DistanceGoalTargets NAMESPACE DistanceGoal:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/DistanceGoalConfig.cmake DESTINATION ${package_dir})

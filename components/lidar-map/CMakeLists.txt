cmake_minimum_required(VERSION 3.16)
project(LidarMap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(NOT TARGET Roboframe::sdk)
    find_package(RoboframeSDK CONFIG REQUIRED)
endif()
add_library(lidar_map STATIC src/recorded_lidar_map_component.cpp)
add_library(LidarMap::component ALIAS lidar_map)
set_target_properties(lidar_map PROPERTIES EXPORT_NAME component)
target_compile_features(lidar_map PUBLIC cxx_std_14)
target_include_directories(lidar_map PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(lidar_map PUBLIC Roboframe::sdk)
roboframe_component_metadata(lidar_map lidar_map)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(lidar_map PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()
if(BUILD_TESTING)
    add_executable(recorded_lidar_map_test tests/recorded_lidar_map_component_test.cpp)
    target_link_libraries(recorded_lidar_map_test PRIVATE LidarMap::component)
    add_test(NAME recorded_lidar_map_test COMMAND recorded_lidar_map_test)
endif()
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/LidarMap)
configure_package_config_file(cmake/LidarMapConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/LidarMapConfig.cmake INSTALL_DESTINATION ${package_dir})
install(TARGETS lidar_map EXPORT LidarMapTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT LidarMapTargets NAMESPACE LidarMap:: DESTINATION ${package_dir})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LidarMapConfig.cmake DESTINATION ${package_dir})

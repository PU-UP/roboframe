cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
include("${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
project(Roboframe VERSION ${ROBOFRAME_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(cmake/Dependencies.cmake)
option(ROBOFRAME_USE_INSTALLED_COMPONENTS "Use installed packages instead of local SDK/components" OFF)

set(ROBOFRAME_DEFAULT_SOURCE SOURCE)
if(ROBOFRAME_USE_INSTALLED_COMPONENTS)
    set(ROBOFRAME_DEFAULT_SOURCE PACKAGE)
endif()
set(ROBOFRAME_SDK_SOURCE "${ROBOFRAME_DEFAULT_SOURCE}" CACHE STRING "SOURCE or PACKAGE for SDK")
set_property(CACHE ROBOFRAME_SDK_SOURCE PROPERTY STRINGS SOURCE PACKAGE)
if(ROBOFRAME_SDK_SOURCE STREQUAL "SOURCE")
    add_subdirectory(sdk)
elseif(ROBOFRAME_SDK_SOURCE STREQUAL "PACKAGE")
    find_package(RoboframeSDK ${ROBOFRAME_REQUIRED_SDK_VERSION} EXACT CONFIG REQUIRED)
else()
    message(FATAL_ERROR "ROBOFRAME_SDK_SOURCE must be SOURCE or PACKAGE")
endif()
get_target_property(selected_sdk Roboframe::sdk ROBOFRAME_SDK_VERSION)
if(NOT selected_sdk VERSION_EQUAL ROBOFRAME_REQUIRED_SDK_VERSION)
    message(FATAL_ERROR "Product selects SDK ${ROBOFRAME_REQUIRED_SDK_VERSION}, found ${selected_sdk}")
endif()
include(cmake/RoboframeComponents.cmake)
roboframe_load_components()

add_library(roboframe_integration
    integration/robot_integration.cpp
    integration/workflow/mapping.cpp
    integration/data_flow/sensor_flow.cpp
    integration/data_flow/pose_buffer.cpp
    integration/data_flow/result_dispatch.cpp)
target_include_directories(roboframe_integration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(roboframe_integration PUBLIC Roboframe::sdk
    PRIVATE Odom::component LidarMap::component MapUpdater::component MapServer::component)
add_library(roboframe_stdio interface/stdio_interface.cpp interface/stdio_channel.cpp)
target_include_directories(roboframe_stdio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(roboframe_stdio PUBLIC Roboframe::sdk)
add_executable(robot_runtime main.cpp)
target_link_libraries(robot_runtime PRIVATE roboframe_stdio roboframe_integration)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(roboframe_integration PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    target_compile_options(roboframe_stdio PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    target_compile_options(robot_runtime PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

    target_link_libraries(shared_map_reader_test PRIVATE Roboframe::sdk)

cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS VERSION ROBOFRAME_VERSION LIMIT_COUNT 1)
project(Roboframe VERSION ${ROBOFRAME_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(ROBOFRAME_USE_INSTALLED_COMPONENTS "Use installed packages instead of local SDK/components" OFF)

set(ROBOFRAME_DEFAULT_SOURCE SOURCE)
if(ROBOFRAME_USE_INSTALLED_COMPONENTS)
    set(ROBOFRAME_DEFAULT_SOURCE PACKAGE)
endif()
set(ROBOFRAME_SDK_SOURCE "${ROBOFRAME_DEFAULT_SOURCE}" CACHE STRING "SOURCE or PACKAGE for SDK")
set_property(CACHE ROBOFRAME_SDK_SOURCE PROPERTY STRINGS SOURCE PACKAGE)
if(ROBOFRAME_SDK_SOURCE STREQUAL "SOURCE")
    add_subdirectory(sdk)
elseif(ROBOFRAME_SDK_SOURCE STREQUAL "PACKAGE")
    find_package(RoboframeSDK 1.0 CONFIG REQUIRED)
else()
    message(FATAL_ERROR "ROBOFRAME_SDK_SOURCE must be SOURCE or PACKAGE")
endif()
include(cmake/RoboframeComponents.cmake)
roboframe_component(sample-accumulator SampleAccumulator 1.0)

add_library(roboframe_integration integration/demo_integration.cpp integration/data_flow.cpp)
target_include_directories(roboframe_integration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(roboframe_integration PUBLIC SampleAccumulator::component)
add_executable(robot_runtime main.cpp interface/stdio_interface.cpp)
target_link_libraries(robot_runtime PRIVATE roboframe_integration)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(roboframe_integration PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    target_compile_options(robot_runtime PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME ipc_check COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/ipc_check.py $<TARGET_FILE:robot_runtime>)
    set_tests_properties(ipc_check PROPERTIES TIMEOUT 15)
endif()

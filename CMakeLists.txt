cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS VERSION ROBOFRAME_VERSION LIMIT_COUNT 1)
project(Roboframe VERSION ${ROBOFRAME_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(ROBOFRAME_BUILD_MAPPING "Build the odometry/scan mapping product" OFF)
option(ROBOFRAME_USE_INSTALLED_COMPONENTS "Use installed packages instead of local SDK/components" OFF)

set(ROBOFRAME_DEFAULT_SOURCE SOURCE)
if(ROBOFRAME_USE_INSTALLED_COMPONENTS)
    set(ROBOFRAME_DEFAULT_SOURCE PACKAGE)
endif()
set(ROBOFRAME_SDK_SOURCE "${ROBOFRAME_DEFAULT_SOURCE}" CACHE STRING "SOURCE or PACKAGE for SDK")
set_property(CACHE ROBOFRAME_SDK_SOURCE PROPERTY STRINGS SOURCE PACKAGE)
if(ROBOFRAME_SDK_SOURCE STREQUAL "SOURCE")
    add_subdirectory(sdk)
elseif(ROBOFRAME_SDK_SOURCE STREQUAL "PACKAGE")
    if(ROBOFRAME_BUILD_MAPPING)
        find_package(RoboframeSDK 1.1 CONFIG REQUIRED)
    else()
        find_package(RoboframeSDK 1.0 CONFIG REQUIRED)
    endif()
else()
    message(FATAL_ERROR "ROBOFRAME_SDK_SOURCE must be SOURCE or PACKAGE")
endif()
include(cmake/RoboframeComponents.cmake)
roboframe_component(sample-accumulator SampleAccumulator 1.0)

add_library(roboframe_integration integration/demo_integration.cpp integration/data_flow.cpp)
target_include_directories(roboframe_integration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(roboframe_integration PUBLIC SampleAccumulator::component)
add_executable(robot_runtime main.cpp interface/stdio_interface.cpp)
target_link_libraries(robot_runtime PRIVATE roboframe_integration)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(roboframe_integration PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    target_compile_options(robot_runtime PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
endif()

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME ipc_check COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/ipc_check.py $<TARGET_FILE:robot_runtime>)
    set_tests_properties(ipc_check PROPERTIES TIMEOUT 15)
endif()

if(ROBOFRAME_BUILD_MAPPING)
    roboframe_component(odom-buffer OdomBuffer 1.0)
    roboframe_component(occupancy-grid OccupancyGrid 1.0)
    roboframe_component(map-store MapStore 1.0)
    add_library(mapping_integration integration/mapping.cpp integration/mapping_data.cpp)
    target_include_directories(mapping_integration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(mapping_integration PUBLIC OdomBuffer::component OccupancyGrid::component MapStore::component)
    add_executable(mapping_runtime mapping_main.cpp interface/mapping_stdio.cpp)
    target_link_libraries(mapping_runtime PRIVATE mapping_integration)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(mapping_integration PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
        target_compile_options(mapping_runtime PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
    if(BUILD_TESTING)
        add_executable(mapping_check tests/mapping_check.cpp)
        target_link_libraries(mapping_check PRIVATE mapping_integration)
        add_test(NAME mapping_check COMMAND mapping_check)
        add_test(NAME mapping_ipc_check COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/mapping_ipc_check.py $<TARGET_FILE:mapping_runtime>)
        add_test(NAME mapping_replay_check COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/mapping_replay_check.py)
        set_tests_properties(mapping_ipc_check PROPERTIES TIMEOUT 30)
    endif()
endif()
